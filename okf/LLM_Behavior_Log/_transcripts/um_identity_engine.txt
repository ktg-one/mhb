\section{Design Intent, Guarantees, Scope}
% -------------------------

\subsection*{Design intent}
UM specifies an identity-maintaining dynamical system that:
(i) integrates symbolic/relational state, memory feedback, affect salience, coercion resistance, and temporal momentum into a single identity update,
(ii) restricts access through a capacity-limited workspace selected by uncertainty- and stake-aware attention,
(iii) broadcasts selected content into identity integration and updates an explicit self-model under a unity constraint,
(iv) closes a perception-action loop and supports learning with temporal credit assignment,
(v) carries an explicit monotone irreversibility stake that modulates both objective and attention.

\subsection*{Guarantees (conditional, audit-grade)}
Under the global assumptions and gain bounds stated in Section~\ref{sec:assumptions}, the Tick$(t)$ map is well-posed on a bounded invariant domain and the identity update remains contractive (no runaway recursion). Under fixed exogenous inputs for a tick, the induced identity update admits a unique fixed point and trajectories remain bounded.

\subsection*{Scope}
UM specifies operational access (capacity-limited workspace, global availability via broadcast into integration, unity-constrained self-model update, and reportable control state). Phenomenology is out of scope; UM is an executable architecture and stability-constrained dynamics, not a theory of subjective qualia.

\section{Metric Structure}
% -------------------------

Let $(\mathcal U,d)$ be a metric space with weighted norm:
\[
d(x,y)=\|x-y\|_W=\sqrt{(x-y)^\top W (x-y)},\qquad W\succ 0.
\]

All non-expansiveness / Lipschitz / contraction claims are with respect to this same metric (or its natural product extensions).

\section{Canonical Identity Dynamics (UM$^2$)}
% -------------------------

The identity engine evolves by:
\[
\boxed{
UM(t)=
\Phi\Big[
\oplus\, f_{\mathcal I}\big(t,S_{\mathcal I}(t),R_{\mathcal I}(t)\big)
+\sum_{\delta\in\mathcal D}\gamma_\delta\, UM(t-\delta)
+\Psi\big(Q(t)\big)
-\Lambda\big(M(t)\big)
+\Omega(\Delta t)
\Big].
}
\]

\subsection*{Identity clause (explicit re-entrance)}
\[
\boxed{
UM(t)\equiv UM\big(UM(t-1)\big)\ \oplus\ \mathrm{Identity}(t).
}
\]

\section{Contractive Requirement and Admissible Realization}
% -------------------------

\subsection*{Contractive requirement}
Assume $\Phi:(\mathcal U,d)\to(\mathcal U,d)$ is a contraction:
\[
d(\Phi(x),\Phi(y)) \le L\, d(x,y),\qquad 0\le L<1,\quad \forall x,y\in\mathcal U.
\]

\subsection*{Admissible realization (contractive stability law)}
A concrete sufficient stability law consistent with the abstract contraction requirement is:
\[
\boxed{
x_{t+1}
=
(1-\gamma)x_t
+
\gamma x_{\text{env},t}
-
\eta \nabla \Phi(x_t),
\qquad
0<\gamma<1,\;\eta>0.
}
\]
For suitable gains $(\gamma,\eta)$ and Lipschitz $\nabla\Phi$ on bounded domains, the induced one-step map is contractive in $d$.
This law is an admissible construction that can be used to implement or approximate $\Phi$ while explicitly maintaining a certified regime.

\section{Global Assumptions (Audit-Ready Consolidation)}
\label{sec:assumptions}
% -------------------------

All closure and stability claims are under A0--A7.

\subsection*{A0. Complete metric + bounded invariant domain}
$(\mathcal U,d)$ is complete and computations are confined to a bounded invariant domain $\mathcal U_B\subset\mathcal U$ (enforced by projection/clipping where needed).

\subsection*{A1. Contractive integration}
$\Phi$ is contractive on $\mathcal U_B$ with constant $L<1$.

\subsection*{A2. Strange-loop and injected operators are bounded/Lipschitz on $\mathcal U_B$}
On $\mathcal U_B$ (and product spaces formed from bounded inputs), $\oplus$, $f_{\mathcal I}$, $\Psi$, $\Lambda$, $\Omega$ are bounded and Lipschitz.

\subsection*{A3. Memory feedback boundedness}
$\gamma_\delta\ge 0$, $\sum_{\delta\in\mathcal D}\gamma_\delta\le 1$, and delayed states retrieved from $\mathcal M(t)$ lie in $\mathcal U_B$.

\subsection*{A4. Workspace capacity and deterministic tie-handling}
Workspace capacity is finite: $\|W(t)\|\le C_W$. If $\mathcal C(t)=\emptyset$ then $W(t)=\emptyset$. If selection ties occur, a fixed deterministic rule breaks ties so $\mathcal A$ is a function.

\subsection*{A5. Selection regularity (analysis-grade)}
$\mathcal A$ is non-expansive with respect to $d$ on bounded inputs (or satisfies a margin/hysteresis condition sufficient to yield effective non-expansiveness). This is a stylized sufficient condition used for analysis; implementations approximate it via smoothing, hysteresis, bounded temperatures, and deterministic tie-breaking with robustness checks.

\subsection*{A6. Broadcast boundedness}
$B:\mathcal W\to\mathcal U$ is bounded and Lipschitz on bounded workspaces:
\[
d(B(W_1),B(W_2)) \le K_B\, d_{\mathcal W}(W_1,W_2),
\quad \|B(W)\|\le b_{\max}.
\]

\subsection*{A7. Stability-preserving learning regime}
Parameters are projected into compact sets:
\[
\Theta(t)\in\mathcal K_\Theta,\qquad \bar\Theta(t)\in\mathcal K_{\bar\Theta},
\]
and all operator Lipschitz constants/gains used in stability bounds are uniformly bounded over $\mathcal K_\Theta$.
Learning rates and projections are chosen so the certified regime inequalities (e.g.\ $L'<1$ below) remain invariant over time.

\subsection*{Implementation enforcement note (non-axiomatic)}
A0--A7 are enforced/approximated by: projection/clipping to $\mathcal U_B$ and $\mathcal K_\Theta$; bounded parameterizations; gradient/activation clipping; norm monitors on injected magnitudes (e.g.\ $\|B(W)\|$, $\|\Psi(Q)\|$, $\|\Lambda(M)\|$); and safe-mode gain reduction (shrink $\beta$, step sizes) if bounds are exceeded.

\section{UM-A: Attention, Uncertainty, and Workspace Selection}
% -------------------------

\subsection*{Pre-tick convention}
$UM(t^-)$ denotes the stored identity at the start of Tick$(t)$ (after prior tick storage; before current attention, broadcast, and integration).

\subsection*{Encoding, prediction, and uncertainty}
\[
z(t) = E_{\Theta}\big(o(t), R_{\mathcal I}(t)\big),\qquad
\hat o(t) = \hat O_{\Theta}\big(z(t), a(t-1)\big),
\]
\[
\varepsilon(t)=\varepsilon\big(o(t),\hat o(t)\big),\qquad
\Sigma(t)=U_{\Theta}\big(\Sigma(t-1),\varepsilon(t),z(t)\big),\quad \Sigma(t)\succeq 0.
\]

\subsection*{Candidates and selection}
\[
\mathcal C(t)=\mathsf{Cand}\big(UM(t^-), S_{\mathcal I}(t), R_{\mathcal I}(t), z(t)\big).
\]

Define a bounded score:
\[
\text{Score}(c;t)=
\alpha_Q\, s_Q(c;Q(t))
+\alpha_\varepsilon\, s_\varepsilon(c;\varepsilon(t))
+\alpha_\Sigma\, s_\Sigma(c;\Sigma(t))
+\alpha_{NF}\, s_{NF}(c;m_{NF}(t))
-\alpha_M\, s_M(c;M(t)),
\quad \alpha_\cdot\ge 0,
\]
with all $s_\cdot$ bounded.

Workspace is selected as:
\[
\boxed{
W(t)=\mathcal A\big(\mathcal C(t); Q(t),\varepsilon(t),\Sigma(t),m_{NF}(t),M(t)\big)
=
\arg\max_{W\subset\mathcal C(t)} \sum_{c\in W}\text{Score}(c;t)
\ \ \text{s.t.}\ \ \|W\|\le C_W.
}
\]

(ii) $B(W(t))$ is injected into identity integration,
(iii) $S_{\text{self}}(t)$ is updated using $W(t)$ under unity constraints,

\section{Contractive Safety of the Broadcast-Augmented Tick Map}
\label{sec:stability}
% -------------------------

\subsection*{Definition: broadcast-augmented map $\Phi'$}
Let $W(\cdot)$ denote the workspace selection induced by $\mathcal A$ and $\mathsf{Cand}$ for fixed tick inputs.
Define the broadcast-augmented identity operator:
\[
\boxed{
\Phi'(x)=\Phi\big(x+\beta\,B(W(x))\big)
}
\]
understood on the bounded domain $\mathcal U_B$ with the same metric $d$.

\subsection*{Small-gain sufficient condition}
Assume the composite injection $x\mapsto \beta B(W(x))$ is Lipschitz on $\mathcal U_B$ with constant $\beta K_{BW}$:
\[
d\!\big(\beta B(W(x)), \beta B(W(y))\big)\le \beta K_{BW}\, d(x,y).
\]
Then:
\[
d(\Phi'(x),\Phi'(y))\le (L+\beta K_{BW})\, d(x,y).
\]
A sufficient condition for contraction is:
\[
\boxed{
L+\beta K_{BW}<1.
}
\]

\subsection*{Theorem (Tick-level contractive safety)}
\textbf{Theorem.}
Assume A0--A7 and suppose $L+\beta K_{BW}<1$. Then, for fixed tick inputs and fixed parameters within the certified regime, the broadcast-augmented identity update in Tick$(t)$ is a contraction on $(\mathcal U_B,d)$ with contraction constant $L'\le L+\beta K_{BW}<1$. Consequently:
\begin{enumerate}
\item the identity update is well-posed on $\mathcal U_B$;
\item for fixed exogenous inputs on the tick, it admits a unique fixed point;
\item repeated application with bounded inputs and enforced projections yields bounded identity trajectories (no runaway recursion).
\end{enumerate}

\paragraph{Proof sketch.}
By A1, $\Phi$ is $L$-contractive. By A5--A6 and bounded-domain composition, the injection $\beta B(W(\cdot))$ is Lipschitz with constant $\beta K_{BW}$ in the same metric $d$. A standard small-gain argument yields contraction constant $L'\le L+\beta K_{BW}<1$ for $\Phi'$. Invariance and boundedness follow from A0--A4 and projection/clipping. \hfill$\square$

\section{Minimal Worked Example (One Tick)}
% -------------------------

This example executes one tick with toy numbers (illustrative, not a claim of optimal parameters).

Let $\mathcal U=\mathbb R$, $d(x,y)=|x-y|$. Choose:
\[
UM(t^-)=0.50,\quad UM(t-1)=0.40,\quad \gamma_1=0.30,\quad Q(t)=0.20,\quad M(t)=0.10,\quad \Delta t=1.
\]
Let injected terms (toy) be:
\[
f_{\mathcal I}=0.60,\quad \Psi(Q)=0.10Q=0.02,\quad \Lambda(M)=0.20M=0.02,\quad \Omega(\Delta t)=0.
\]
Let workspace selection pick $W(t)=\{c^\star\}$ with broadcast $B(W)=0.50$ and gain $\beta=0.10$ so $\beta B(W)=0.05$.
Let $\oplus$ be addition and $\Phi(x)=\tanh(x)$.

Compute pre-$\Phi$ argument:
\[
x = f_{\mathcal I}+\gamma_1 UM(t-1)+\Psi(Q)-\Lambda(M)+\Omega(\Delta t)+\beta B(W)
=0.60+0.30(0.40)+0.02-0.02+0+0.05
=0.77.
\]
Then:
\[
UM(t)=\tanh(0.77)\approx 0.647.
\]
Let $\Delta m_{NF}(t)=0.02$ so $m_{NF}(t+1)=m_{NF}(t)+0.02$ and the objective $\mathcal J(t)$ updates accordingly.
